var TAB='Store',CHUNK=45000;
function tab_(){var s=SpreadsheetApp.getActiveSpreadsheet(),t=s.getSheetByName(TAB);if(!t){t=s.insertSheet(TAB);t.getRange(1,1,1,3).setValues([['key','part','json']]).setFontWeight('bold');t.setFrozenRows(1);t.getRange('A:C').setNumberFormat('@')}return t}
function readAll_(prefix){var t=tab_(),n=t.getLastRow(),o={};if(n<2)return o;var v=t.getRange(2,1,n-1,3).getValues(),parts={};
  v.forEach(function(r){var k=String(r[0]);if(k.indexOf(prefix)!==0)return;(parts[k]=parts[k]||[]).push([+r[1],String(r[2])])});
  for(var k in parts){parts[k].sort(function(a,b){return a[0]-b[0]});try{o[k]=JSON.parse(parts[k].map(function(p){return p[1]}).join(''))}catch(e){}}return o}
function putKey_(key,data){var t=tab_(),n=t.getLastRow();
  if(n>1){var ks=t.getRange(2,1,n-1,1).getValues(),keep=[],i;for(i=0;i<ks.length;i++)if(String(ks[i][0])!==key)keep.push(i+2);
    if(keep.length!==ks.length){var all=t.getRange(2,1,n-1,3).getValues().filter(function(r){return String(r[0])!==key});t.getRange(2,1,n-1,3).clearContent();if(all.length)t.getRange(2,1,all.length,3).setValues(all)}}
  var s=JSON.stringify(data),rows=[],p=0;for(var j=0;j<s.length;j+=CHUNK)rows.push([key,String(p++),s.substr(j,CHUNK)]);
  if(!rows.length)rows.push([key,'0','null']);var at=t.getLastRow()+1;t.getRange(at,1,rows.length,3).setNumberFormat('@').setValues(rows)}
function delKey_(key){var t=tab_(),n=t.getLastRow();if(n<2)return;var all=t.getRange(2,1,n-1,3).getValues(),keep=all.filter(function(r){return String(r[0])!==key});
  t.getRange(2,1,n-1,3).clearContent();if(keep.length)t.getRange(2,1,keep.length,3).setValues(keep)}
function json_(o){return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON)}
function okKey_(k){var p=k.indexOf(':');if(p<0)return false;var pre=k.substring(0,p),rest=k.substring(p+1);
  if(pre!=='pub'&&pre!=='adm')return false;if(rest.length<1||rest.length>40)return false;
  var ok='ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789_.-';
  for(var i=0;i<rest.length;i++)if(ok.indexOf(rest.charAt(i))<0)return false;return true}
var CK='nimex-pub';
function cacheGet_(){var c=CacheService.getScriptCache(),n=c.get(CK+'-n');if(!n)return null;var keys=[];for(var i=0;i<+n;i++)keys.push(CK+'-'+i);var all=c.getAll(keys),b='';for(i=0;i<keys.length;i++){if(!all[keys[i]])return null;b+=all[keys[i]]}return Utilities.ungzip(Utilities.newBlob(Utilities.base64Decode(b),'application/x-gzip')).getDataAsString('UTF-8')}
function cachePut_(s){var b=Utilities.base64Encode(Utilities.gzip(Utilities.newBlob(s,'text/plain')).getBytes()),o={},n=0;for(var i=0;i<b.length;i+=90000)o[CK+'-'+(n++)]=b.substr(i,90000);if(n<=90){o[CK+'-n']=String(n);CacheService.getScriptCache().putAll(o,21600)}}
function cacheClear_(){CacheService.getScriptCache().remove(CK+'-n')}
function doGet(){var s=null;try{s=cacheGet_()}catch(e){}
  if(!s){var k=readAll_('pub:'),c=k['pub:cfg'],vp=(c&&c.viewerPages)||[];
    var a=readAll_('adm:');for(var x in a)k[x]=a[x];
    s=JSON.stringify({ok:true,keys:k});try{cachePut_(s)}catch(e){}}
  return ContentService.createTextOutput(s).setMimeType(ContentService.MimeType.JSON)}
function doPost(e){var b;try{b=JSON.parse(e.postData.contents)}catch(x){return json_({ok:false,error:'bad_request'})}
  var key=PropertiesService.getScriptProperties().getProperty('ADMIN_KEY');if(!key||!b.key||b.key!==key){Utilities.sleep(1200);return json_({ok:false,error:'denied'})}
  if(b.a==='auth')return json_({ok:true});
  if(b.a==='admin')return json_({ok:true,keys:readAll_('adm:')});
  var lock=LockService.getScriptLock();if(!lock.tryLock(25000))return json_({ok:false,error:'busy'});
  try{
    if(b.a==='put'){var k=String(b.k||'');if(!okKey_(k))return json_({ok:false,error:'bad_key'});putKey_(k,b.data);cacheClear_();return json_({ok:true})}
    if(b.a==='del'){var k2=String(b.k||'');if(!okKey_(k2))return json_({ok:false,error:'bad_key'});delKey_(k2);cacheClear_();return json_({ok:true})}
    return json_({ok:false,error:'unknown'})
  }finally{lock.releaseLock()}}
